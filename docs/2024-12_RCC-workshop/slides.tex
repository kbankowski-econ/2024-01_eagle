\begin{frame}
  \frametitle{Evaluating Full Compliance with Fiscal Rules}
  \framesubtitle{What We Do in This Project}
  
  \vspace{-1ex}
  
  \begin{itemize}
    \item We assess whether macroeconomic outcomes in the euro area would differ with full adherence to the fiscal framework.
    
    \item We construct counterfactual scenarios using macroeconomic models:
      \begin{itemize}
          \item Estimating the fiscal adjustments needed for full SGP compliance using simple country-specific models.
          \item Simulating alternative macroeconomic trajectories with a large-scale model for the euro area.
      \end{itemize}
    
    \item Full compliance with the SGP would require significant early fiscal adjustments in EMU countries.
    
    \item Adhering to the SGP would moderate early 2000s growth and reduce the need for post-GFC fiscal consolidation, which aggravated the second downturn.
    
    \item Sound fiscal positions would prevent the debt accumulation that was observed.
    
    \item Our results hold with both ex-post and real-time cyclical estimates.
  \end{itemize}

\end{frame}

\section{Context \& Literature}
\begin{frame}
  \frametitle{Context \& Related Literature}
  
  \textbf{Context}
  \begin{itemize}
    \item Major reform of EU fiscal rules took effect on 30 April 2024.
    \item Now is the final chance to draw lessons from the past to improve the new framework's success.
  \end{itemize}
  
  \textbf{Related Literature}
  \begin{itemize}
    \item Benefits: fiscal discipline, lower borrowing costs, reduced volatility (\textcolor{royalblue}{Potrafke (2023)}; \textcolor{royalblue}{Brändle et al. (2023)}).
    \item However, SGP largely failed to meet its goals (\textcolor{royalblue}{Haroutunian et al. (2024)}).
    \item Failures due to design flaws, noted by \textcolor{royalblue}{Eyraud \& Wu (2015)}; \textcolor{royalblue}{Kamps \& Leiner-Killinger (2019)}.
    \item Weak enforcement also hindered success (\textcolor{royalblue}{Larch et al. (2023)}).
  \end{itemize}
  
  \textbf{Our Contribution}
  \begin{itemize}
    \item We analyse the role of compliance in the SGP's effectiveness.
  \end{itemize}

\end{frame}

\section{Fiscal adjustments}
\begin{frame}
  \frametitle{Country-Specific Models}
  \framesubtitle{Behavioural Equations}
  
  \begin{itemize}
    \item \textbf{Structural balance:}
    \begin{equation*}
      SBB_{t} = F\left(SBB_{t-1}, YEG_{t-1}, MAL_{t-1}, INP_{t}\right) + e_{t}^{SBB}
    \end{equation*}

    \item \textbf{Real GDP growth:}
    \begin{equation*}
      YER_{t} = 0.5YER_{t-1} + 0.5YET_{t} - 0.55\Delta SPB_{t} - 0.4YEG_{t-1} + e_{t}^{YER}
    \end{equation*}

    \item \textbf{Cyclical component of the budget balance:}
    \begin{equation*}
      CYC_{t} = \epsilon YEG_{t} + e_{t}^{CYC}
    \end{equation*}
  \end{itemize}
  
\end{frame}

\begin{frame}
  \frametitle{Country-Specific Models}
  \framesubtitle{Identities}

    \begin{itemize}
      \item Primary budget balance: $PBB_{t}=SPB_{t}+CYC_{t}+OFF_{t}$ 
      \item Interest payments: $INP_{t}=\left(\frac{IIR_{t}}{100}\frac{MAL_{t-1}}{100}\frac{1}{1+\nicefrac{YEN_{t}}{100}}\right)\times100$
      \item Overall budget balance: $B9N_{t}=PBB_{t}-INP_{t}$
      \item Debt accumulation: $MAL_{t}=\frac{MAL_{t-1}}{1+\nicefrac{YEN_{t}}{100}}-B9N_{t}+SFA_{t}$
      \item Structural primary balance: $SPB_{t}=SBB_{t}+INP_{t}$
      \item Nominal output growth: $YEN_{t}=\left[\left(1+\frac{YER_{t}}{100}\right)\left(1+\frac{YED_{t}}{100}\right)-1\right]\times100$
      \item Output gap: $YEG_{t}=\left[\left(1+\frac{YEG_{t-1}}{100}\right)\frac{1+\nicefrac{YER_{t}}{100}}{1+\nicefrac{YET_{t}}{100}}-1\right]\times100$
  
      \end{itemize}

\end{frame}

\section{Conclusions}
\begin{frame}
  \frametitle{Conclusions}
  \begin{itemize}
      \item Our paper does not attempt to defend the pre-reformed SGP. However, we emphasize that compliance issues significantly contributed to the SGP's failure to achieve its objectives.
      \item Had the fiscal rules been fully adhered to and enforced, the euro area could have achieved more favourable fiscal outcomes (i.e., smoothing the cycle and building fiscal buffers).
      \item Critical lesson for the future: the success of the newly reformed framework will depend on adherence to its provisions from the outset.
  \end{itemize}
\end{frame}
